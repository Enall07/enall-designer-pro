// Generate Image using Gemini 3.1 Flash Image API
async function generateImage() {
const promptText = document.getElementById('promptInput').value.trim();
const activeImages = uploadedImages.filter(img => img !== null);

if (!promptText && activeImages.length === 0) {
alert("Harap masukkan setidaknya Deskripsi Visual atau unggah minimal 1 Foto Referensi Inspirasi Karakter terlebih dahulu!");
return;
}

document.getElementById('placeholderState').classList.add('hidden');
document.getElementById('resultContainer').classList.add('hidden');
document.getElementById('loadingState').classList.remove('hidden');
document.getElementById('generateBtn').disabled = true;

let finalPrompt = promptText || "Professional high quality character portrait photography";
if (selectedPose) finalPrompt += `. Pose: ${selectedPose}`;
if (selectedAngle) finalPrompt += `. Camera Angle: ${selectedAngle}`;
if (selectedStyle) finalPrompt += `. Photographic Style: ${selectedStyle}`;
finalPrompt += `. Masterpiece, ultra detailed, professional digital art.`;

try {
let parts = [{ text: finalPrompt }];

activeImages.forEach((imgData, index) => {
const base64Data = imgData.split(',')[1];
const mimeType = imgData.split(';')[0].split(':')[1];
parts.push({
inlineData: {
mimeType: mimeType,
data: base64Data
}
});
});

if (activeImages.length > 0) {
parts[0].text = `Combine and transform these ${activeImages.length} character reference image(s) following this precise instruction: ${finalPrompt}`;
}

const payload = {
contents: [{ role: 'user', parts: parts }],
generationConfig: {
responseModalities: ['IMAGE'],
imageConfig: { aspectRatio: selectedRatio }
}
};

const apiKey = "";
const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3.1-flash-image:generateContent?key=${apiKey}`;

let response = null;
let retries = 3;
let delay = 1000;

for (let i = 0; i < retries; i++) {
try {
response = await fetch(apiUrl, {
method: 'POST',
headers: { 'Content-Type': 'application/json' },
body: JSON.stringify(payload)
});
if (response.ok) break;
} catch (err) {
if (i === retries - 1) throw err;
}
await new Promise(res => setTimeout(res, delay));
delay *= 2;
}

if (!response || !response.ok) {
throw new Error("Gagal terhubung ke server AI Gemini.");
}

const result = await response.json();
const part = result?.candidates?.[0]?.content?.parts?.find(p => p.inlineData);

if (!part) {
throw new Error("AI tidak dapat menghasilkan gambar. Coba sesuaikan prompt Anda.");
}

generatedImageUrl = `data:${part.inlineData.mimeType};base64,${part.inlineData.data}`;

const outputImg = document.getElementById('outputImage');
outputImg.src = generatedImageUrl;
outputImg.className = `max-w-full h-auto object-contain transition-all ${currentFilter === 'normal' ? '' : currentFilter}`;

document.getElementById('loadingState').classList.add('hidden');
document.getElementById('resultContainer').classList.remove('hidden');
document.getElementById('downloadBtn').disabled = false;
document.getElementById('downloadBtn').className = "bg-blue-600 hover:bg-blue-700 text-white font-bold py-2.5 px-5 rounded-xl text-xs flex items-center space-x-2 shadow-lg shadow-blue-500/20 transition cursor-pointer";

} catch (error) {
console.error(error);
alert("Terjadi kesalahan saat memproses gambar: " + error.message);
document.getElementById('loadingState').classList.add('hidden');
document.getElementById('placeholderState').classList.remove('hidden');
} finally {
document.getElementById('generateBtn').disabled = false;
}
}

// Save / Download Image with Watermark baked in
function downloadImage() {
if (!generatedImageUrl) return;

const canvas = document.createElement('canvas');
const ctx = canvas.getContext('2d');
const img = new Image();

img.onload = function() {
canvas.width = img.naturalWidth;
canvas.height = img.naturalHeight;

ctx.filter = getCanvasFilterString(currentFilter);
ctx.drawImage(img, 0, 0);
ctx.filter = 'none';

const padding = Math.max(20, canvas.width * 0.02);
const fontSize = Math.max(14, Math.floor(canvas.height * 0.03));
ctx.font = `bold ${fontSize}px 'Plus Jakarta Sans', sans-serif`;

const text = "ENALL DESIGNER PRO";
const textMetrics = ctx.measureText(text);
const badgeSize = fontSize * 1.2;

ctx.fillStyle = 'rgba(0, 0, 0, 0.5)';
ctx.beginPath();
const boxX = padding;
const boxY = canvas.height - padding - fontSize - 16;
const boxWidth = textMetrics.width + badgeSize + 28;
const boxHeight = fontSize + 16;
const radius = 8;

ctx.roundRect(boxX, boxY, boxWidth, boxHeight, radius);
ctx.fill();

ctx.fillStyle = 'rgba(255, 255, 255, 0.95)';
ctx.fillText(text, boxX + 14, boxY + fontSize + 2);

const circleX = boxX + 14 + textMetrics.width + 12;
const circleY = boxY + (boxHeight / 2);
ctx.fillStyle = '#2563eb';
ctx.beginPath();
ctx.arc(circleX, circleY, badgeSize * 0.45, 0, 2 * Math.PI);
ctx.fill();

ctx.strokeStyle = '#ffffff';
ctx.lineWidth = Math.max(2, badgeSize * 0.08);
ctx.beginPath();
ctx.moveTo(circleX - 3, circleY);
ctx.lineTo(circleX - 1, circleY + 3);
ctx.lineTo(circleX + 4, circleY - 2);
ctx.stroke();

const link = document.createElement('a');
link.download = `EnallDesignerPro_${Date.now()}.jpg`;
link.href = canvas.toDataURL('image/jpeg', 0.92);
link.click();
};
img.src = generatedImageUrl;
}

function getCanvasFilterString(filterClass) {
switch(filterClass) {
case 'filter-cinematic': return 'contrast(120%) saturate(110%) brightness(95%)';
case 'filter-vintage': return 'sepia(50%) hue-rotate(-30deg) saturate(140%)';
case 'filter-warm': return 'sepia(30%) saturate(130%) hue-rotate(-10deg)';
case 'filter-cool': return 'hue-rotate(180deg) saturate(90%) brightness(105%)';
case 'filter-brightness': return 'brightness(1.2)';
case 'filter-contrast': return 'contrast(1.3)';
case 'filter-grayscale': return 'grayscale(100%)';
default: return 'none';
}
}
